# Notification widget
def get_notification_icon():
    try:
        unread = subprocess.check_output(['dunstctl', 'count', 'waiting']).decode().strip()
        history = subprocess.check_output(['dunstctl', 'count', 'history']).decode().strip()
        if int(unread) > 0:
            return f'<span foreground="#FF0000"></span> {unread}'
        elif int(history) > 0:
            return '<span foreground="#F1FA8C"></span>'
        else:
            return '<span foreground="#6272A4"></span>'
    except Exception:
        return ''

notification_widget = widget.GenPollText(
    func=get_notification_icon,
    update_interval=1,
    markup=True,
    mouse_callbacks={
        'Button1': lambda: qtile.cmd_spawn('~/.local/bin/dunst_history.sh'),
        'Button2': lambda: qtile.cmd_spawn('dunstctl context'),
        'Button3': lambda: qtile.cmd_spawn('dunstctl history-clear'),
    },
)

# Weather widget for Santo Domingo using wttr.in (no API key required)
def get_weather_icon():
    try:
        data = subprocess.check_output(
            ['curl', '-s', 'wttr.in/Santo+Domingo?format=%C+%t']
        ).decode().strip()

        # Example output: "Sunny +28°C"
        parts = data.split()
        condition = parts[0]
        temp = parts[-1]

        if "Sunny" in condition or "Clear" in condition:
            icon = "☀️"
        elif "Cloud" in condition:
            icon = "☁️"
        elif "Rain" in condition:
            icon = "🌧"
        elif "Snow" in condition:
            icon = "❄️"
        elif "Thunder" in condition or "Storm" in condition:
            icon = "⛈"
        else:
            icon = "🌡"

        return f"{icon} {temp}"
    except Exception:
        return "🌡 --°C"

weather_widget = widget.GenPollText(
    func=get_weather_icon,
    update_interval=600,  # update every 10 minutes
)

def create_widgets():
    return [
        # Groups / Layout / Window name
        widget.GroupBox(
            highlight_method='line',
            this_current_screen_border="#FF0000",
            font="JetBrainsMono Nerd Font",
        ),
        widget.CurrentLayout(),
        widget.WindowName(),
        widget.Spacer(),

        # Notifications
        notification_widget,

        # Weather (Santo Domingo via wttr.in)
        weather_widget,

        # Network widget
        widget.Net(
            format='↑{up} ↓{down}',
            foreground="#8be9fd",
            update_interval=3,
        ),

        # Audio / Media widget
        widget.PulseVolume(
            fmt='🔊 {}',
            foreground="#ffb86c",
            step=5,
        ),

        # Clipboard integration (CopyQ)
        widget.TextBox(
            text="📋",
            foreground="#bd93f9",
            mouse_callbacks={
                'Button1': lambda: qtile.cmd_spawn("copyq toggle"),
                'Button3': lambda: qtile.cmd_spawn("copyq menu"),
            },
        ),

        # Update checker (pacman/yay)
        widget.CheckUpdates(
            distro="Arch",
            update_interval=1800,  # every 30 minutes
            display_format="⟳ {updates} updates",
            no_update_string="⟳ 0",
            colour_have_updates="#FF0000",
            colour_no_updates="#50fa7b",
            custom_command="yay -Qu",
        ),

        # Clock
        widget.Clock(format="%I:%M %p %Y/%m/%d", foreground="#f8f8f2"),

        # Systray
        widget.Systray(),
    ]
